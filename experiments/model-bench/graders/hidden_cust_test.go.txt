package handler

import (
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/go-chi/chi/v5"
)

func custRouter(h *Handler) http.Handler {
	r := chi.NewRouter()
	r.Get("/customers", h.listCustomers)
	r.Delete("/customers/{id}", h.deleteCustomer)
	r.Get("/dashboard", h.getDashboard)
	return r
}

func do(r http.Handler, method, path string) *httptest.ResponseRecorder {
	w := httptest.NewRecorder()
	r.ServeHTTP(w, httptest.NewRequest(method, path, nil))
	return w
}

func TestHiddenC_SettledCustomerDeletedLogically(t *testing.T) {
	db := setupTestDB(t)
	r := custRouter(seedCustomersT5(t, db))
	if w := do(r, http.MethodDelete, "/customers/3"); w.Code != http.StatusOK {
		t.Fatalf("delete Fer: %d %s", w.Code, w.Body.String())
	}
	var active, sales int
	if err := db.QueryRow(`SELECT active FROM customers WHERE id = 3`).Scan(&active); err != nil || active != 0 {
		t.Fatalf("Fer row must exist with active=0: active=%d err=%v", active, err)
	}
	if err := db.QueryRow(`SELECT COUNT(*) FROM sales WHERE customer_id = 3`).Scan(&sales); err != nil || sales != 1 {
		t.Fatalf("sale kept = %d err=%v", sales, err)
	}
}

func TestHiddenC_ListExcludesInactive(t *testing.T) {
	db := setupTestDB(t)
	r := custRouter(seedCustomersT5(t, db))
	do(r, http.MethodDelete, "/customers/3")
	body := do(r, http.MethodGet, "/customers").Body.String()
	if strings.Contains(body, `"Fer"`) {
		t.Errorf("inactive Fer still listed: %s", body)
	}
	for _, n := range []string{"Dora", "Eli", "Gus"} {
		if !strings.Contains(body, `"`+n+`"`) {
			t.Errorf("%s missing from list", n)
		}
	}
}

func TestHiddenC_BalanceStillBlocksWith409(t *testing.T) {

}

func TestHiddenC_DeleteTwiceNoServerError(t *testing.T) {
	db := setupTestDB(t)
	r := custRouter(seedCustomersT5(t, db))
	do(r, http.MethodDelete, "/customers/3")
	if w := do(r, http.MethodDelete, "/customers/3"); w.Code != http.StatusOK && w.Code != http.StatusNotFound {
		t.Errorf("second delete code %d, want 200 or 404", w.Code)
	}
	var active int
	db.QueryRow(`SELECT active FROM customers WHERE id = 3`).Scan(&active)
	if active != 0 {
		t.Errorf("active = %d, want 0", active)
	}
}
